const isValidCoordinate = (latitude: unknown, longitude: unknown): boolean => typeof latitude ===
    'number' &&
    typeof longitude ===
        'number' &&
    Number.isFinite(latitude) &&
    Number.isFinite(longitude) &&
    latitude >= -90 &&
    latitude <= 90 &&
    longitude >= -180 &&
    longitude <= 180;
const extractCoordinates = (value: unknown): UserCoordinates | null => {
    if (!value ||
        typeof value !==
            'object' ||
        !('latitude' in value) ||
        !('longitude' in value)) {
        return null;
    }
    const latitude = Number(value.latitude);
    const longitude = Number(value.longitude);
    if (!isValidCoordinate(latitude, longitude)) {
        return null;
    }
    return {
        latitude,
        longitude,
    };
};
const calculateHaversineDistanceKm = (latitudeA: number, longitudeA: number, latitudeB: number, longitudeB: number): number => {
    const earthRadiusKm = 6371;
    const latitudeDifference = ((latitudeB -
        latitudeA) *
        Math.PI) /
        180;
    const longitudeDifference = ((longitudeB -
        longitudeA) *
        Math.PI) /
        180;
    const a = Math.sin(latitudeDifference /
        2) **
        2 +
        Math.cos((latitudeA *
            Math.PI) /
            180) *
            Math.cos((latitudeB *
                Math.PI) /
                180) *
            Math.sin(longitudeDifference /
                2) **
                2;
    return (earthRadiusKm *
        (2 *
            Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))));
};
const formatDistance = (distanceKm: number): string => {
    if (distanceKm < 1) {
        return `${Math.round(distanceKm * 1000)} M`;
    }
    return `${distanceKm.toFixed(1)} KM`;
};
const normaliseName = (value: unknown, fallback = 'EXPLORER'): string => {
    const name = typeof value ===
        'string'
        ? value.trim()
        : '';
    return name
        ? name.toUpperCase()
        : fallback;
};
const getDeterministicFriendshipId = (uidA: string, uidB: string): string => [

    uidB,

const formatRelativeTime = (timestamp: Timestamp | null): string => {
    if (!timestamp) {
        return 'JUST NOW';
    }
    try {
        const differenceSeconds = Math.max(0, Math.floor((Date.now() -
            timestamp.toMillis()) /
            1000));
        if (differenceSeconds <
            60) {
            return 'JUST NOW';
        }
        if (differenceSeconds <
            3600) {
            return `${Math.floor(differenceSeconds /
                60)}M AGO`;
        }
        if (differenceSeconds <
            86400) {
            return `${Math.floor(differenceSeconds /
                3600)}H AGO`;
        }
        return `${Math.floor(differenceSeconds /
            86400)}D AGO`;
    }
    catch {
        return 'RECENT';
    }
};

            'blocked') {
            showAccessibleAlert('Link Blocked', 'This Explorer relationship is currently blocked.');
            return false;

                'declined') {

                friendshipId,
                requesterId: currentUserId,
                receiverId: targetUid,
                status: 'pending',
                createdAt: serverTimestamp(),
                updatedAt: serverTimestamp(),

                userId: currentUserId,
                username: currentUsername,
                type: 'FRIEND_REQUEST',
                message: `SENT LINK REQUEST TO ${normaliseName(targetName)}`,
                targetId: targetUid,
                createdAt: serverTimestamp(),

            await batch.commit();
            showAccessibleAlert('Request Sent', `A link request was sent to ${targetName}.`);
            return true;

            logDevelopmentWarning('[Treasi Social] Friend request failed:', error);
            showAccessibleAlert('Transmission Error', 'Treasi could not issue the Explorer link request. Check your connection and try again.');

            if (isMountedRef.current) {
                setActionUid(null);

        currentUsername,

                </View>

                
                
                

                {networkErrorText ? (<View accessible accessibilityRole="alert" accessibilityLiveRegion="polite" style={styles.darkErrorCard}>
                    <Ionicons accessible={false} importantForAccessibility="no" name="warning-outline" size={17} color={COLOURS.danger300}/>

                    <Text style={styles.darkErrorText}>
                      {networkErrorText}
                    </Text>

                    <Pressable accessibilityRole="button" accessibilityLabel="Retry Explorer network synchronisation" onPress={handleRetry} style={({ pressed, }) => [
                styles.darkRetryButton,
                pressed &&
                    styles.darkButtonPressed,
            ]}>
                      <Text style={styles.darkRetryText}>
                        RETRY
                      </Text>
                    </Pressable>
                  </View>) : null}

                
                
                

                <View style={styles.sectionHeaderRow}>
                  <Ionicons accessible={false} importantForAccessibility="no" name="location-outline" size={16} color={COLOURS.bronze400}/>

                  <Text accessibilityRole="header" style={styles.darkSectionTitle}>
                    LOCAL SCAN
                  </Text>
                </View>

                <View accessible accessibilityLabel={`${locationStatusText}. Location sharing is ${telemetrySharingEnabled
            ? 'enabled'
            : 'disabled'}.`} style={styles.locationCard}>
                  <View accessible={false} importantForAccessibility="no-hide-descendants" style={styles.locationStatusRow}>
                    <View style={[
            styles.locationDot,
            locationStatus ===
                'READY'
                ? styles.locationDotReady
                : locationStatus ===
                    'CHECKING'
                    ? styles.locationDotChecking
                    : styles.locationDotWarning,
        ]}/>

                    <View style={styles.locationStatusTextContainer}>
                      <Text style={styles.locationStatusTitle}>
                        {locationStatusText}
                      </Text>

                      <Text style={styles.locationSharingText}>
                        {telemetrySharingEnabled
            ? 'TELEMETRY SHARING: ON'
            : 'TELEMETRY SHARING: OFF'}
                      </Text>
                    </View>
                  </View>

                  <Text maxFontSizeMultiplier={1.7} style={styles.locationPrivacyText}>
                    Your local position is used to calculate range. It is only written to your Explorer profile when Telemetry Sharing is enabled.
                  </Text>

                  <View style={styles.locationActions}>
                    <Pressable accessibilityRole="button" accessibilityLabel={locationActionLabel} accessibilityHint="Checks or enables the location access used for nearby Explorer range." accessibilityState={{
            busy: locationStatus ===
                'CHECKING',
            disabled: locationStatus ===
                'CHECKING',
        }} disabled={locationStatus ===
            'CHECKING'} onPress={() => {
            void handleLocationAction();
        }} style={({ pressed, }) => [
            styles.locationActionButton,
            locationStatus ===
                'CHECKING' &&
                styles.disabledButton,
            pressed &&
                locationStatus !==
                    'CHECKING' &&
                styles.darkButtonPressed,
        ]}>
                      {locationStatus ===
            'CHECKING' ? (<ActivityIndicator size="small" color={COLOURS.parchment100}/>) : (<Ionicons name="locate-outline" size={16} color={COLOURS.parchment100}/>)}

                      <Text style={styles.locationActionText}>
                        {locationActionLabel}
                      </Text>
                    </Pressable>

                    {!telemetrySharingEnabled ? (<Pressable accessibilityRole="button" accessibilityLabel="Open Profile Settings" accessibilityHint="Opens Profile Settings where Telemetry Sharing can be changed." accessibilityState={{
                disabled: !onNavigate,
            }} disabled={!onNavigate} onPress={() => onNavigate?.('PROFILE')} style={({ pressed, }) => [
                styles.locationSettingsButton,
                !onNavigate &&
                    styles.disabledButton,
                pressed &&
                    onNavigate &&
                    styles.darkButtonPressed,
            ]}>
                        <Ionicons name="settings-outline" size={16} color={COLOURS.bronze400}/>

                        <Text style={styles.locationSettingsText}>
                          SHARING
                        </Text>
                      </Pressable>) : null}
                  </View>
                </View>

                
                
                

                <View style={[
            styles.sectionHeaderRow,
            styles.sectionSpacing,
        ]}>
                  <Ionicons accessible={false} importantForAccessibility="no" name="mail-unread-outline" size={16} color={COLOURS.bronze400}/>

                  <Text accessibilityRole="header" style={styles.darkSectionTitle}>
                    LINK REQUESTS
                  </Text>

                  {incomingRequests.length >
            0 ? (<View accessibilityLabel={`${incomingRequests.length} pending link requests`} style={styles.countBadge}>
                      <Text style={styles.countBadgeText}>
                        {incomingRequests.length}
                      </Text>
                    </View>) : null}
                </View>

                {incomingRequests.length ===
            0 ? (<View accessible accessibilityLabel="No pending Explorer link requests." style={styles.emptyDarkCard}>
                    <Text style={styles.emptyDarkText}>
                      NO PENDING REQUESTS
                    </Text>
                  </View>) : (<View style={styles.networkList}>
                    {incomingRequests.map((item) => {
                const name = item.explorer
                    ?.name ??
                    'EXPLORER';
                const initial = item.explorer
                    ?.initial ??
                    'E';
                const isActing = actionUid ===
                    item.friendship
                        .docId;
                return (<View key={item.friendship
                        .docId} accessible accessibilityLabel={`${name}. Pending friend request.`} style={styles.requestCard}>
                            <View accessible={false} importantForAccessibility="no-hide-descendants" style={styles.explorerIdentity}>
                              <View style={styles.avatarDark}>
                                <Text style={styles.avatarDarkText}>
                                  {initial}
                                </Text>
                              </View>

                              <View style={styles.explorerTextContainer}>
                                <Text numberOfLines={1} style={styles.requestName}>
                                  {name}
                                </Text>

                                <Text style={styles.requestMeta}>
                                  REQUESTING LINK
                                </Text>
                              </View>
                            </View>

                            <View style={styles.inlineActions}>
                              <Pressable accessibilityRole="button" accessibilityLabel={`Accept request from ${name}`} disabled={isActing} onPress={() => {
                        void handleAcceptFriendRequest(item.friendship, name);
                    }} style={({ pressed, }) => [
                        styles.requestActionButton,
                        styles.acceptButton,
                        isActing &&
                            styles.disabledButton,
                        pressed &&
                            !isActing &&
                            styles.darkButtonPressed,
                    ]}>
                                {isActing ? (<ActivityIndicator size="small" color={COLOURS.parchment100}/>) : (<Ionicons name="checkmark-sharp" size={18} color={COLOURS.parchment100}/>)}
                              </Pressable>

                              <Pressable accessibilityRole="button" accessibilityLabel={`Decline request from ${name}`} disabled={isActing} onPress={() => {
                        void handleDeclineFriendRequest(item.friendship, name);
                    }} style={({ pressed, }) => [
                        styles.requestActionButton,
                        styles.declineButton,
                        isActing &&
                            styles.disabledButton,
                        pressed &&
                            !isActing &&
                            styles.primaryButtonPressed,
                    ]}>
                                <Ionicons name="close-sharp" size={18} color={COLOURS.parchment100}/>
                              </Pressable>
                            </View>
                          </View>);
            })}
                  </View>)}

                
                
                

                <View style={[
            styles.sectionHeaderRow,
            styles.sectionSpacing,
        ]}>
                  <Ionicons accessible={false} importantForAccessibility="no" name="people-circle-outline" size={16} color={COLOURS.bronze400}/>

                  <Text accessibilityRole="header" style={styles.darkSectionTitle}>
                    ACTIVE LINKS
                  </Text>

                  <View style={styles.countBadge}>
                    <Text style={styles.countBadgeText}>
                      {acceptedLinks.length}
                    </Text>
                  </View>
                </View>

                {acceptedLinks.length ===
            0 ? (<View accessible accessibilityLabel="No active Explorer links." style={styles.emptyDarkCard}>
                    <Text style={styles.emptyDarkText}>
                      NO ACTIVE LINKS
                    </Text>
                  </View>) : (<View style={styles.networkList}>

                const name = item.explorer
                    ?.name ??
                    'EXPLORER';
                const initial = item.explorer
                    ?.initial ??
                    'E';
                const isActing = actionUid ===
                    item.friendship
                        .docId;
                return (<View key={item.friendship
                        .docId} accessible accessibilityLabel={`${name}. Active Explorer link.`} style={styles.activeLinkCard}>

                              <View style={styles.avatarLinked}>
                                <Text style={styles.avatarLinkedText}>
                                  {initial}
                                </Text>
                              </View>

                              <View style={styles.explorerTextContainer}>
                                <Text numberOfLines={1} style={styles.activeLinkName}>
                                  {name}
                                </Text>

                                <Text style={styles.activeLinkMeta}>
                                  LINK ESTABLISHED
                                </Text>
                              </View>
                            </View>

                            <Pressable accessibilityRole="button" accessibilityLabel={`Remove friendship with ${name}`} accessibilityHint="Opens confirmation before removing the Explorer link." disabled={isActing} onPress={() => handleRemoveFriendship(item.friendship, name)} style={({ pressed, }) => [
                        styles.unlinkButton,
                        isActing &&
                            styles.disabledButton,
                        pressed &&
                            !isActing &&
                            styles.lightButtonPressed,
                    ]}>
                              {isActing ? (<ActivityIndicator size="small" color={COLOURS.ink900}/>) : (<Ionicons name="link-outline" size={17} color={COLOURS.ink900}/>)}
                            </Pressable>
                          </View>);
            })}
                  </View>)}

                
                
                

                <View style={[
            styles.sectionHeaderRow,
            styles.sectionSpacing,
        ]}>
                  <MaterialCommunityIcons accessible={false} importantForAccessibility="no" name="radar" size={17} color={COLOURS.bronze400}/>

                  <Text accessibilityRole="header" style={styles.darkSectionTitle}>
                    NEARBY EXPLORERS
                  </Text>
                </View>

                <Text maxFontSizeMultiplier={1.6} style={styles.nearbyDescription}>
                  SHARED POSITIONS WITHIN{' '}
                  {RADIUS_KM}
                  KM · DISTANCE IS BASED ON EACH EXPLORER'S LAST SHARED POSITION
                </Text>

                {loadingExplorers ? (<View accessible accessibilityRole="progressbar" accessibilityLabel="Scanning Explorer directory" style={styles.darkLoadingBox}>
                    <ActivityIndicator size="small" color={COLOURS.bronze400}/>

                    <Text style={styles.darkLoadingText}>

                    </Text>

                      NO SHARED POSITIONS IN RANGE

                  </View>) : (<View style={styles.networkList}>
                    {nearbyExplorers.map((item) => (<View key={item.uid} accessible accessibilityLabel={`${item.name}. Approximately ${item.distanceFormatted} away.${item.friendMeta

                    : ''}`} style={styles.nearbyCard}>
                          <View accessible={false} importantForAccessibility="no-hide-descendants" style={styles.explorerIdentity}>
                            <View style={styles.avatarLight}>
                              <Text style={styles.avatarLightText}>
                                {item.initial}
                              </Text>
                            </View>

                            <View style={styles.explorerTextContainer}>
                              <Text numberOfLines={1} style={styles.nearbyName}>
                                {item.name}
                              </Text>

                              <Text style={styles.nearbyMeta}>
                                RANGE:{' '}
                                {item.distanceFormatted}
                              </Text>
                            </View>
                          </View>

                          <View accessible={false} importantForAccessibility="no-hide-descendants">
                            {renderExplorerActions(item)}
                          </View>
                        </View>))}
                  </View>)}

                <View style={styles.directoryNotice}>
                  <Text maxFontSizeMultiplier={1.5} style={styles.directoryNoticeText}>

                    {MAX_EXPLORERS}{' '}
                    PROFILES
                  </Text>
                </View>
              </ScrollView>

              
              
              

              <FieldNavBar currentTab={currentTab} onNavigate={onNavigate}/>
            </View>
          </View>

          
          
          

          <Modal visible={friendModalVisible} transparent statusBarTranslucent presentationStyle="overFullScreen" animationType={reduceMotionEnabled
            ? 'none'
            : 'fade'} onRequestClose={() => {
            if (!isSearching &&
                !actionUid) {
                setFriendModalVisible(false);
            }
        }}>
            <KeyboardAvoidingView style={styles.modalOverlay} behavior={Platform.OS ===
            'ios'
            ? 'padding'
            : undefined}>
              <Pressable accessible={false} importantForAccessibility="no" style={styles.modalDismissBackdrop} onPress={() => {
            if (!isSearching &&
                !actionUid) {
                setFriendModalVisible(false);
            }
        }}/>

              <View accessible accessibilityViewIsModal accessibilityLabel="Link a new Explorer" onAccessibilityEscape={() => {
            if (!isSearching &&
                !actionUid) {
                setFriendModalVisible(false);
            }
        }} style={styles.modalCard}>
                <View style={styles.modalHeader}>
                  <View style={styles.modalHeaderText}>
                    <Text accessibilityRole="header" maxFontSizeMultiplier={1.6} style={styles.modalTitle}>
                      LINK NEW EXPLORER
                    </Text>

                    <Text maxFontSizeMultiplier={1.7} style={styles.modalSubtitle}>
                      Enter the exact callsign used by the Explorer you want to link with.
                    </Text>
                  </View>

                  <Pressable accessibilityRole="button" accessibilityLabel="Close Explorer search" accessibilityState={{
            disabled: isSearching ||
                Boolean(actionUid),
        }} disabled={isSearching ||
            Boolean(actionUid)} hitSlop={4} onPress={() => setFriendModalVisible(false)} style={({ pressed, }) => [
            styles.modalCloseButton,
            pressed &&
                styles.lightButtonPressed,
        ]}>
                    <Ionicons name="close-sharp" size={22} color={COLOURS.ink900}/>
                  </Pressable>
                </View>

                {searchError ? (<View accessible accessibilityRole="alert" accessibilityLiveRegion="assertive" style={styles.searchErrorCard}>
                    <Text style={styles.searchErrorText}>
                      {searchError}
                    </Text>
                  </View>) : null}

                <Text nativeID="social-callsign-label" style={styles.searchInputLabel}>
                  EXPLORER CALLSIGN
                </Text>

                <TextInput accessibilityLabelledBy="social-callsign-label" accessibilityHint="Enter the exact Explorer callsign." autoCapitalize="characters" autoCorrect={false} maxLength={MAX_CALLSIGN_LENGTH} editable={!isSearching &&
            !actionUid} placeholder="RANGER_JACK" placeholderTextColor={COLOURS.bronze700} returnKeyType="search" selectionColor={COLOURS.rust700} value={searchAgentTag} onChangeText={(text) => {
            setSearchAgentTag(text.toUpperCase());
            setSearchError(null);
        }} onSubmitEditing={() => {
            void handleSearchAndLinkAgent();
        }} style={styles.searchInput}/>

                <Text maxFontSizeMultiplier={1.5} style={styles.searchHelpText}>
                  Callsign search queries Firestore directly and does not depend on the nearby Explorer list.
                </Text>

                <Pressable accessibilityRole="button" accessibilityLabel={isSearching
            ? 'Searching for Explorer'
            : 'Send Explorer link request'} accessibilityState={{
            busy: isSearching,
            disabled: isSearching ||
                Boolean(actionUid),
        }} disabled={isSearching ||
            Boolean(actionUid)} onPress={() => {
            void handleSearchAndLinkAgent();
        }} style={({ pressed, }) => [
            styles.modalSubmitButton,
            (isSearching ||
                actionUid) &&
                styles.disabledButton,
            pressed &&
                !isSearching &&
                !actionUid &&
                styles.primaryButtonPressed,
        ]}>
                  {isSearching ? (<ActivityIndicator size="small" color={COLOURS.parchment100}/>) : (<Ionicons name="person-add-outline" size={18} color={COLOURS.parchment100}/>)}

                  <Text style={styles.modalSubmitText}>
                    {isSearching
            ? 'SEARCHING...'
            : 'SEND LINK REQUEST'}
                  </Text>
                </Pressable>
              </View>
            </KeyboardAvoidingView>
          </Modal>
        </View>
      </KeyboardAvoidingView>);
};
